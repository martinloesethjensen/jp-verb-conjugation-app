# Grammar Building Blocks: Milestone 1 (Foundations) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Every grammar point says which building block (slot) it attaches to and what follows, the app builds that pattern for any verb, adjective or noun, and a Building blocks screen teaches the blocks; ので and 〜ほうがいいです ship as the first points in the new format.

**Architecture:** Adjective and noun forms are generated by `scripts/update_data.py` into a new signed `data/words.json`, exactly like the generated verb forms. Grammar attachment rules gain `slots`, `da_to_na` and `then`; VerbKit maps a slot to a form id, applies the one な rule, and appends the ending. A `WordsStore` (a copy of the `FuriganaStore` pattern) syncs the word list. The app adds a Building blocks screen and rebuilds the grammar card's "How it attaches" section from the slots.

**Tech Stack:** Python 3 standard library (data pipeline), Swift 5 mode, SwiftUI (iOS 26), SwiftData, XcodeGen, XCTest, `unittest`.

**Spec:** `docs/superpowers/specs/2026-10-06-grammar-building-blocks-design.md` (milestone 1 only).

**Base:** `main` with the spec merged. Branch: `claude/grammar-building-blocks`.

## Global Constraints

- Python tests: `cd scripts && python3 -m unittest -v test_update_data test_form_catalogue` (all must pass). Package tests: `swift test --package-path Packages/VerbKit`; commit only when the **exit code is 0**.
- After any data edit run `python3 scripts/update_data.py`, then `python3 scripts/update_data.py --check` must print "data is up to date". **Every `data/` change needs the owner to run `python3 scripts/sign_manifest.py sign` before merge**; never commit a signature yourself, say in the PR that it needs signing.
- After adding app files run `./scripts/generate-project.sh`. iOS build: `xcodebuild -project JPVerbConjugation.xcodeproj -scheme JPVerbConjugation_iOS -destination 'id=<simulator UDID>' -derivedDataPath <scratch>/dd build`. **iPad and macOS checks are parked:** do not build the macOS scheme.
- Slot names (exact): `plain`, `plainNeg`, `plainPast`, `plainPastNeg`, `stem`, `te`. Form ids: `short_pos`, `short_neg`, `short_past`, `short_past_neg`, `stem`, `te`.
- Word classes (exact raw values): `verb`, `i-adjective`, `na-adjective`, `noun`.
- いい-type adjectives are an explicit list, `いい` and `かっこいい`. Never "anything ending in いい" (かわいい is regular: かわいくない).
- The な switch: only on slot `plain`, only for `na-adjective` and `noun`: drop a final だ, add な.
- All new content is written by the assistant and marked unreviewed in the PR; no textbook text is copied.
- Commit messages end with the trailer from the session's attribution reminder.

```python
# --- Adjectives and nouns (words.json) ---------------------------------------

WORD_FILE_CLASSES = ("i-adjective", "na-adjective", "noun")
# いい and its compounds conjugate from よい. An explicit list: かわいい also ends in いい
# but is regular (かわいくない).
II_ADJECTIVES = {"いい", "かっこいい"}


def word_forms(word):
    """The generated forms of an adjective or noun, in catalogue order. Raises
    ValueError for a class words.json does not hold or a malformed い-adjective."""
    cls, d = word.get("class"), word.get("dict", "")
    if cls == "i-adjective":
        if not d.endswith("い") or len(d) < 2:
            raise ValueError(f"{d}: an い-adjective must end in い")
        base = d[:-2] + "よ" if d in II_ADJECTIVES else d[:-1]
        forms = {"short_pos": d, "short_neg": base + "くない", "short_past": base + "かった",
                 "short_past_neg": base + "くなかった", "stem": base, "te": base + "くて"}
    elif cls in ("na-adjective", "noun"):
        if not d:
            raise ValueError("a word needs a dict form")
        forms = {"short_pos": d + "だ", "short_neg": d + "じゃない", "short_past": d + "だった",
                 "short_past_neg": d + "じゃなかった", "te": d + "で"}
        if cls == "na-adjective":
            forms["stem"] = d
    else:
        raise ValueError(f"{d}: class must be one of {WORD_FILE_CLASSES}, got {cls!r}")
    return {key: forms[key] for key in form_catalogue.ids(cls) if key in forms}


def apply_word_forms(words_doc):
    """Set every word's `forms` (the script owns them). Returns True if anything changed."""
    changed = False
    for word in words_doc["words"]:
        wanted = word_forms(word)
        if word.get("forms") != wanted:
            word["forms"] = wanted
            changed = True
    return changed


def check_words(words_doc, readings):
    """Problems with the hand-written fields of words.json."""
    problems, seen = [], set()
    for word in words_doc.get("words", []):
        d = word.get("dict", "<no dict>")
        key = (word.get("class"), d)
        if key in seen:
            problems.append(f"{d}: duplicate word")
        seen.add(key)
        if word.get("class") not in WORD_FILE_CLASSES:
            problems.append(f"{d}: class must be one of {WORD_FILE_CLASSES}")
        if not isinstance(word.get("meaning"), str) or not word["meaning"]:
            problems.append(f"{d}: meaning missing or empty")
        if "jlpt" in word and word["jlpt"] not in JLPT_LEVELS:
            problems.append(f"{d}: jlpt must be one of {sorted(JLPT_LEVELS)}")
        kanji = word.get("kanji")
        if kanji:
            got = kana_reading(kanji, readings)
            # い-adjective kanji include the okurigana (高い); compare whole spellings.
            if got != d:
                problems.append(f"{d}: kanji {kanji} reads as {got}, expected {d}")
    return problems


def dump_words(doc):
    return json.dumps(doc, ensure_ascii=False, indent=2) + "\n"
```

```swift
import Foundation

/// A building block a grammar pattern attaches to.
public enum GrammarSlot: String, Codable, CaseIterable, Sendable {
    case plain, plainNeg, plainPast, plainPastNeg, stem, te

    /// The form that holds this slot for every word class.
    public var formID: FormID {
        switch self {
        case .plain: "short_pos"
        case .plainNeg: "short_neg"
        case .plainPast: "short_past"
        case .plainPastNeg: "short_past_neg"
        case .stem: "stem"
        case .te: "te"
        }
    }
}

/// Anything with a word class and forms by id: a curated verb or a word from words.json.
public protocol SlotSource {
    var wordClass: WordClass { get }
    var dict: String { get }
    func slotForm(_ id: FormID) -> String?
}

extension Word: SlotSource {
    public func slotForm(_ id: FormID) -> String? { forms[id] }
}

extension Verb: SlotSource {
    public var wordClass: WordClass { .verb }
    public func slotForm(_ id: FormID) -> String? {
        switch id.rawValue {
        case "short_pos": forms.shortPos
        case "short_neg": forms.shortNeg
        case "short_past": forms.shortPast
        case "short_past_neg": forms.shortPastNeg
        case "te": forms.te
        case "stem": forms.stemOrDerived
        default: nil
        }
    }
}

public extension SlotSource {
    /// The word's form for `slot`. With `daToNa`, a な-adjective's or noun's plain form
    /// drops だ and takes な (しずかな, あめな); nothing else changes.
    func form(for slot: GrammarSlot, daToNa: Bool = false) -> String? {
        guard let form = slotForm(slot.formID) else { return nil }
        if daToNa, slot == .plain, wordClass == .naAdjective || wordClass == .noun, form.hasSuffix("だ") {
            return String(form.dropLast()) + "な"
        }
        return form
    }
}
```

```swift
import SwiftUI
import VerbKit

/// "How it attaches": per rule, the slot chips, the ending and a live example. "Try another
/// word" steps through the words of that class. Rules without slots show their text as before.
struct AttachmentSection: View {
    @Environment(VerbStore.self) private var verbStore
    @Environment(WordsStore.self) private var wordsStore
    let rules: [AttachmentRule]
    @State private var offset = 0

    private func words(_ cls: WordClass) -> [SlotSource] {
        cls == .verb ? verbStore.verbs : wordsStore.words(of: cls)
    }

    private func example(_ rule: AttachmentRule) -> (word: String, built: String)? {
        let pool = words(rule.wordClass)
        guard !pool.isEmpty, rule.then != nil else { return nil }
        for step in 0..<pool.count {
            let word = pool[(offset + step) % pool.count]
            for slot in rule.slots {
                if let built = rule.build(word, slot: slot), let base = word.form(for: slot) {
                    return (base, built)
                }
            }
        }
        return nil
    }

    var body: some View {
        VStack(alignment: .leading, spacing: 12) {
            HStack {
                Text("How it attaches").font(.title3.weight(.semibold))
                Spacer()
                if rules.contains(where: { $0.then != nil }) {
                    Button("Try another word", systemImage: "arrow.triangle.2.circlepath") { offset += 1 }
                        .font(.footnote)
                }
            }
            ForEach(Array(rules.enumerated()), id: \.offset) { _, rule in
                VStack(alignment: .leading, spacing: 8) {
                    HStack(spacing: 6) {
                        Text(rule.wordClass.displayName).font(.subheadline.weight(.semibold))
                        if let condition = rule.condition {
                            JapaneseText("· \(condition)").font(.subheadline).foregroundStyle(.secondary)
```
